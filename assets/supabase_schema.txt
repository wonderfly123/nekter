  sf_account_id text NOT NULL,
  health_status text NOT NULL CHECK (health_status = ANY (ARRAY['Healthy'::text, 'At Risk'::text, 'Critical'::text])),
  trend text CHECK (trend = ANY (ARRAY['Improving'::text, 'Declining'::text, 'Stable'::text])),
  created_at timestamp with time zone DEFAULT now(),
  health_score numeric CHECK (health_score >= 0::numeric AND health_score <= 100::numeric),

  CONSTRAINT account_health_history_sf_account_id_fkey FOREIGN KEY (sf_account_id) REFERENCES public.accounts(sf_account_id)
);
